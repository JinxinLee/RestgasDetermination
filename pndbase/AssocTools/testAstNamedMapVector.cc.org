#include "Panda/Panda.hh"
#include "AssocTools/AstTestTools.hh"
#include "AssocTools/AstNamedMapVector.hh"
#include "ErrLogger/ErrLog.hh"

#include <string>
#include "BbrStdUtils/String.hh"

int main() {
  AstTestBanner("AstNamedMapVector");

  std::string *aWord = new std::string("Radiohead");
  std::string *aWord2 = new std::string("OK Computer");
  std::string *aWord3 = new std::string("Decon Blue");
  std::string *aWord4 = new std::string("Raintown");
  
  AstNamedMapVector<std::string, std::string> anNM("MyMap", &babar::String::rwHash, &babar::String::rwHash);
  
  anNM.append(aWord, aWord2);
  anNM.append(aWord3, aWord4);

  ErrMsg(routine) << "Map has name " << anNM.name() << endmsg;

  std::vector<std::string*> list1(anNM.list1());
  std::vector<std::string*> list2(anNM.list2());
  for (size_t wombat = 0;wombat < list1.size() ; wombat++){
    ErrMsg(routine) << *list1[wombat] << " " << *list2[wombat] << endmsg;
  }

  const std::string *foundWord = anNM.find1(aWord);
  if (foundWord !=0){
    ErrMsg(routine)<< *aWord << " wrote " << *foundWord << endmsg;
  } else {
    ErrMsg(warning)<< "Failed to find " << *aWord << " in map." << endmsg;
  }
  
  ErrMsg(routine)<< "Map contains " << anNM.members() << " entries." << endmsg;
  
  if (anNM.append(aWord, aWord4)) {
    ErrMsg(warning)<< "Ooops. Successfully apended " << *aWord << " and " 
		   << *aWord4 << " to map." << endmsg;
  } else {
    ErrMsg(routine)<< "Correctly failed to add " << *aWord << " and " 
		   << *aWord4 << " to map." << endmsg;
  }
  
  
  anNM.clear();
  
  ErrMsg(routine) << "After clearing, map contains " 
		  << anNM.members() << " entries. " << endmsg;
  
}
