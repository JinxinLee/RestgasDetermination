void failed(int i, const string& stmp) {
  cerr << "Failed at " << i << " : " ;
  cerr << stmp << endl;
}

int main () {

  // Starting tests
  cout << "Starting tests ... " << endl;

  string str1;
  // now for some gibberish
  str1 += "Now is the time for all\n";
  str1 += "good men to     act\n";
  str1 += "get Together and add\n";
  str1 += "1+1";

    boost::RegEx reg1("a[lc]+");
  if (regExSearch(str1,reg1) != "all") failed (1,regExSearch(str1,reg1));

  boost::RegEx reg2(".\\+.");
  if (regExSearch(str1,reg2) != "1+1") failed (2,regExSearch(str1,reg2));

  boost::RegEx reg3("[P-Z][a-z]*");
  if (regExSearch(str1,reg3) != "Together") failed (3,regExSearch(str1,reg3));

  boost::RegEx reg4("[^a-km-z]$");
  if (regExSearch(str1,reg4) != "l") failed (4,regExSearch(str1,reg4));

  boost::RegEx reg5("^[^aeiouAEIOU]");
  if (regExSearch(str1,reg5) != "N") failed (5,regExSearch(str1,reg5));

  boost::RegEx reg6("ac?t");
  if (regExSearch(str1,reg6) != "act") failed (6,regExSearch(str1,reg6));

  string str2;
  // now for some more gibberish
  str2 += "RegEx will find the first expression in the line\n";
  str2 += "whereas RogueWave will find the last\n";
  str2 += "expression in the buffer";

  boost::RegEx reg100("expression in the [a-z]+$");
  if (regExSearch(str2,reg100) != "expression in the line") 
    failed (100,regExSearch(str2,reg100));

  boost::RegEx reg101("expression");
  if (regExIndex(str2,reg101) != 26) failed (101,regExSearch(str2,reg101));

  boost::RegEx reg102("expresson");
  if (regExIndex(str2,reg102) != boost::RegEx::npos) failed (102,regExSearch(str2,reg102));
 

  if (regExIndex(str2,reg101,27) != 86) failed (103,regExSearch(str2,reg101,27));

  boost::RegEx reg103("expression in the [a-z]+");
  if (regExSearch(str2,reg103,27) != "expression in the buffer") 
      failed (104,regExSearch(str2,reg103,27));

  boost::RegEx reg104("notfound");
  if (regExSearch(str2,reg104,27) != "") 
      failed (105,regExSearch(str2,reg104,27));

  if (regExIndex(str2,reg104,20) != boost::RegEx::npos) 
    failed (106,regExSearch(str2,reg104,20));
 
  // Finished tests
  cout << "Tests completed." << endl;

  return 0;

}
